# Examples usage:
# Linux:
#   cmake -B build -G Ninja \
#     -DCMAKE_EXPORT_COMPILE_COMMANDS=ON \
#     -DCMAKE_BUILD_TYPE=Release \
#     -DAYAKA_TARGET_DEVICE=cuda
#   cmake --build build --config Release
# Windows (Ninja + LLVM/Clang, cl.exe only for CUDA nvcc):
#   CUDAHOSTCXX=cl.exe cmake -B build -G Ninja \
#     -DCMAKE_C_COMPILER=clang-cl \
#     -DCMAKE_CXX_COMPILER=clang-cl \
#     -DCMAKE_LINKER=lld-link \
#     -DCMAKE_RC_COMPILER=llvm-rc \
#     -DCMAKE_MT=llvm-mt \
#     -DCMAKE_EXPORT_COMPILE_COMMANDS=ON \
#     -DCMAKE_BUILD_TYPE=Release \
#     -DAYAKA_TARGET_DEVICE=cuda
#   cmake --build build --config Release

cmake_minimum_required(VERSION 3.26)
project(ayaka_kernel LANGUAGES CXX)

# CMake
set(CMAKE_COLOR_DIAGNOSTICS ON)
set(CMAKE_VERBOSE_MAKEFILE ON CACHE BOOL "Verbose makefile output")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_SHARED_LIBRARY_PREFIX "")

# Target device backend (cuda|rocm|cpu)
set(AYAKA_TARGET_DEVICE "cuda" CACHE STRING "Target device backend for ayaka (cuda|rocm|cpu)")
message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")
message(STATUS "Target device: ${AYAKA_TARGET_DEVICE}")

# CXX 
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC AND AYAKA_TARGET_DEVICE STREQUAL "cuda")
  # Ensure MSVC can find CUDA toolkit headers
  set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -I${CUDAToolkit_INCLUDE_DIRS}")
endif()

# Utils 
include(${CMAKE_CURRENT_LIST_DIR}/cmake/utils.cmake)
# include(FetchContent)

 Python
find_package(Python COMPONENTS Interpreter Development.Module ${SKBUILD_SABI_COMPONENT} REQUIRED)

# Torch
append_cmake_prefix_path("torch" "torch.utils.cmake_prefix_path")
find_package(Torch REQUIRED)
clear_cuda_arches(CMAKE_FLAG)

# torch_python is not exported by TorchConfig.cmake; find it manually
get_filename_component(TORCH_LIB_DIR "${TORCH_LIBRARY}" DIRECTORY)
find_library(TORCH_PYTHON_LIB torch_python PATHS "${TORCH_LIB_DIR}" NO_DEFAULT_PATH)
if(NOT TORCH_PYTHON_LIB)
  message(FATAL_ERROR "torch_python not found in ${TORCH_LIB_DIR}")
endif()
message(STATUS "Found torch_python: ${TORCH_PYTHON_LIB}")

# CUDA / ROCm
if(AYAKA_TARGET_DEVICE STREQUAL "cuda")
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set_property(GLOBAL PROPERTY CUDA_SEPARABLE_COMPILATION ON)
  message(STATUS "Detected CUDA_VERSION=${CUDA_VERSION}")
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)

  # Supported NVIDIA architectures
  set(CUDA_SUPPORTED_ARCHS "7.5;8.0;8.6;8.7;8.9;9.0;10.0")

  # Extract architectures from TORCH_CUDA_ARCH_LIST if set
  if(DEFINED TORCH_CUDA_ARCH_LIST AND NOT TORCH_CUDA_ARCH_LIST STREQUAL "")
    clear_cuda_arches(CUDA_ARCH_FLAGS)
    extract_unique_cuda_archs_ascending(CUDA_ARCHS "${CUDA_ARCH_FLAGS}")
    cuda_archs_loose_intersection(CUDA_ARCHS "${CUDA_SUPPORTED_ARCHS}" "${CUDA_ARCHS}")
  else()
    set(CUDA_ARCHS "${CUDA_SUPPORTED_ARCHS}")
  endif()
  message(STATUS "CUDA target architectures: ${CUDA_ARCHS}")
elseif(AYAKA_TARGET_DEVICE STREQUAL "rocm")
  enable_language(HIP)
  message(STATUS "ROCm/HIP build enabled")
  # TODO: Add ROCm architecture detection and library discovery
elseif(AYAKA_TARGET_DEVICE STREQUAL "cpu")
  message(STATUS "CPU-only build (no GPU kernels)")
  set(AYAKA_COMPILE_FLAGS "-DAYAKA_NO_CUDA_RUNTIME")
else()
  message(FATAL_ERROR "Unknown AYAKA_TARGET_DEVICE: ${AYAKA_TARGET_DEVICE}")
endif()

 Pybind11
append_cmake_prefix_path("pybind11" "pybind11.get_cmake_dir()")
find_package(pybind11 CONFIG REQUIRED)

# Build
set(AYAKA_EXTENSION_LIBS ${TORCH_LIBRARIES} ${TORCH_PYTHON_LIB})
if(AYAKA_TARGET_DEVICE STREQUAL "cuda")
  list(APPEND AYAKA_EXTENSION_LIBS CUDA::cudart)
endif()

define_extension_target(
  NAME _C
  DESTINATION ayaka
  LANGUAGE CXX
  SOURCES csrc/torch_bindings.cpp
  COMPILE_FLAGS ${AYAKA_COMPILE_FLAGS}
  INCLUDE_DIRECTORIES
    ${PROJECT_SOURCE_DIR}/include
  LIBRARIES ${AYAKA_EXTENSION_LIBS}
)
